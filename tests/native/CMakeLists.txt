cmake_minimum_required(VERSION 3.16)
project(limivia_native C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(FIRMWARE "${CMAKE_CURRENT_SOURCE_DIR}/../../main")
set(CJSON "${CMAKE_CURRENT_SOURCE_DIR}/../../managed_components/espressif__cjson/cJSON")
set(UNITY "${IDF_PATH}/components/unity/unity/src")
if(NOT EXISTS "${CJSON}/cJSON.c" OR NOT EXISTS "${UNITY}/unity.c")
    message(FATAL_ERROR "先用 ESP-IDF 6.0.2 reconfigure 解析锁定组件，并传入 -DIDF_PATH")
endif()
add_executable(test_v3 test_v3.c "${FIRMWARE}/engine.c" "${FIRMWARE}/rpc.c" "${FIRMWARE}/flows.c"
    "${FIRMWARE}/operations.c" "${FIRMWARE}/resources.c" "${FIRMWARE}/json.c"
    "${CJSON}/cJSON.c" "${UNITY}/unity.c")
add_executable(test_transport test_transport.c "${FIRMWARE}/framing.c" "${FIRMWARE}/tx.c" "${UNITY}/unity.c")
option(SANITIZE "ASan and UBSan" ON)
foreach(target test_v3 test_transport)
    target_include_directories(${target} PRIVATE "${FIRMWARE}" "${CJSON}" "${UNITY}")
    target_compile_definitions(${target} PRIVATE CJSON_NESTING_LIMIT=16)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -g -fno-omit-frame-pointer)
    target_link_libraries(${target} PRIVATE m)
    if(SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-pie)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -no-pie)
    endif()
endforeach()
add_test(NAME firmware_v3_contract COMMAND test_v3)
add_test(NAME firmware_transport COMMAND test_transport)
