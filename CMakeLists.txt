cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32c3)
set(LIMIVIA_PROFILE "bus" CACHE STRING "bus, signal or iot")
if(NOT LIMIVIA_PROFILE MATCHES "^(bus|signal|iot)$")
    message(FATAL_ERROR "LIMIVIA_PROFILE must be bus, signal or iot")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_SOURCE_DIR}/sdkconfig.defaults;${CMAKE_SOURCE_DIR}/profiles/${LIMIVIA_PROFILE}.defaults")
set(PROJECT_VER "0.3.0a7")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(limivia-${LIMIVIA_PROFILE}-v3rpc)
